const VIDEO_CHUNKS=['00','01','02','03','04','05','06','07'];
let teacherBlobUrl=null;
async function loadTeacherVideo(){
  videoState.textContent='加载老师视频…';
  try{
    const parts=await Promise.all(VIDEO_CHUNKS.map(async n=>{
      const r=await fetch(`video/${n}.txt`,{cache:'no-store'});
      if(!r.ok) throw new Error(`video/${n}.txt HTTP ${r.status}`);
      return (await r.text()).trim();
    }));
    const b64=parts.join('');
    const bin=atob(b64);
    const bytes=new Uint8Array(bin.length);
    for(let i=0;i<bin.length;i++) bytes[i]=bin.charCodeAt(i);
    if(teacherBlobUrl) URL.revokeObjectURL(teacherBlobUrl);
    teacherBlobUrl=URL.createObjectURL(new Blob([bytes],{type:'video/mp4'}));
    teacher.src=teacherBlobUrl;
    teacher.load();
  }catch(e){
    videoError.style.display='grid';
    videoError.textContent='老师视频加载失败：'+e.message;
    videoState.textContent='错误';
    console.error(e);
  }
}
