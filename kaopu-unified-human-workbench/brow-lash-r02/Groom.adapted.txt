import {grow,followLash} from '../brow-lash-r02/Growth.mjs';
import{BoundSurfaceSampler}from'./BoundSurfaceSampler.mjs';
import{THREE,makeFiberMaterial,makeFiberGeometry,writeStrand,commitGeometry}from'./FiberMaterial.mjs';
import{SurfaceChart,headFrame,random,curve2,cubic,visibleRim,bindOffsets,sampleOffsets,anchorFrame,add,sub,mul,dot,unit,mix,clamp,smooth}from'./SurfaceBinding.mjs';
export const SCHEMA='kaopu/brow-lash-groom@1';
// Exact CommonViewer.update contract: native (x,y,z) -> display (x,z,-y).
export const nativeToDisplay=p=>[p[0],p[2],-p[1]];
export const DEFAULTS=Object.freeze({enabled:true,brows:true,lashes:true,seed:7219,browDensity:1,browWidth:1,browLength:1,browArch:0,lashDensity:1,lashLength:1,lashCurl:1,fiberWidth:1,roughness:.48,color:'#503322'});
export const PRESETS={natural:{label:'自然深棕',note:'轻疏眉头 · 短长交错',values:{}},fine:{label:'清淡细眉',note:'透出皮肤 · 更细的毛梢',values:{browDensity:.68,browWidth:.8,browLength:.87,lashDensity:.84,lashLength:.9,fiberWidth:.85,color:'#73513c',roughness:.55}},strong:{label:'浓眉短睫',note:'保留分束 · 不画实心边',values:{browDensity:1.22,browWidth:1.1,browLength:1.05,lashLength:.8,lashCurl:.86,color:'#38271f',roughness:.5}},soft:{label:'柔弯长睫',note:'分层弯曲 · 非齐刷刷扇面',values:{browDensity:.88,browWidth:.85,browArch:.3,lashDensity:1.07,lashLength:1.19,lashCurl:1.15,color:'#59402d',roughness:.43}}};
export function validate(input={}){if(!input||typeof input!=='object'||Array.isArray(input))throw Error('眉睫配方必须为对象');const s={...DEFAULTS,...input};for(const[k,v]of Object.entries(s)){if(!(k in DEFAULTS))throw Error('未知眉睫参数 '+k);if(['enabled','brows','lashes'].includes(k)){if(typeof v!=='boolean')throw Error('无效开关 '+k);}else if(k==='color'){if(!/^#[0-9a-f]{6}$/i.test(v))throw Error('无效毛发颜色');}else if(k==='seed'){if(!Number.isInteger(v)||v<0||v>2147483647)throw Error('无效种子');}else if(k==='browArch'){if(!Number.isFinite(v)||Math.abs(v)>1)throw Error('无效眉弓增量');}else if(k==='roughness'){if(!Number.isFinite(v)||v<.2||v>.8)throw Error('粗糙度越界');}else if(!Number.isFinite(v)||v<.45||v>1.6)throw Error('眉睫参数越界 '+k);}return s;}
const pointCount=13;
export class BrowLashGroom{
 constructor({model,viewer,controller}){this.model=model;this.landmark=i=>nativeToDisplay(model.eyeSurface.landmark(i));this.viewer=viewer;this.controller=controller;this.settings=validate();this.group=new THREE.Group();this.group.name='KAOPU_BROW_LASH_R01';viewer.mesh.add(this.group);this.batches=[];this.revision=0;this.updateCount=0;this.errors=[];this.ready=false;this.visible=true;this.lastSignature='';this.selected='natural';this.lastGeometryMS=0;this.maxRootOffsetMM=0;this.originalRender=viewer.render.bind(viewer);const self=this;viewer.render=function(...args){if(self.ready)self.sync();return self.originalRender(...args);};
 this.listener=()=>{clearTimeout(this.editTimer);this.editTimer=setTimeout(()=>{if(!this.disposed){try{this.build();viewer.render();}catch(e){this.errors.push(String(e));this.ready=false;}}},80);};controller.addEventListener('change',this.listener);
 const archive=model.archive.bind(model),restore=model.restore.bind(model);this.originalArchive=archive;this.originalRestore=restore;model.archive=()=>({...archive(),browLash:{schema:SCHEMA,settings:{...this.settings},selected:this.selected}});model.restore=a=>{const p=a?.browLash;if(p&&p.schema!==SCHEMA)throw Error('眉睫配方版本不兼容');const next=validate(p?.settings||{}),old=this.settings;try{const out=restore(a);this.settings=next;this.selected=p?.selected||'natural';this.listener();return out;}catch(e){this.settings=old;throw e;}};this.build();}
 charts(){const g=this.viewer.mesh.geometry,f=headFrame({eyeSurface:{landmark:this.landmark}}),eye=this.model.eyeSurface,mask=new Uint8Array(this.model.vertexCount);for(const e of eye.eyes)for(const i of e.vertices)mask[i]=1;this.frame=f;const source=this.model.positions,nativeDisplay=new Float32Array(source.length);for(let i=0;i<source.length;i+=3){nativeDisplay[i]=source[i];nativeDisplay[i+1]=source[i+2];nativeDisplay[i+2]=-source[i+1];}this.display=new SurfaceChart(g.attributes.position.array,g.index.array,f,{normals:g.attributes.normal.array});this.skinChart=new SurfaceChart(nativeDisplay,this.model.faces,f,{mask:eye.skin});this.globeChart=new SurfaceChart(nativeDisplay,this.model.faces,f,{mask});this.chartGeometry=g;this.nativeGeometrySignature=this.model.headMorphology?.revision||0;}
 build(){const start=performance.now();this.ready=false;const g=this.viewer.mesh.geometry;if(!g?.index)throw Error('需要原总台已渲染的索引表皮');this.charts();for(const b of this.batches){b.mesh.geometry.dispose();b.mesh.material.userData.depth.dispose();b.mesh.material.dispose();this.group.remove(b.mesh);}this.batches=[];this.omitted={brow:0,upper:0,lower:0};this.rimRows=[];const s=this.settings,f=this.frame,aa=!!this.viewer.renderer.getContext().getContextAttributes()?.antialias;this.msaa=aa;
 const grouped=grow(this,g);
 for(const[kind,strands]of Object.entries(grouped)){if(!strands.length)throw Error('没有生成 '+kind+'；不能用空几何报告成功');const geometry=makeFiberGeometry(strands),material=makeFiberMaterial({roughness:s.roughness,anisotropy:.82,aa}),mesh=new THREE.Mesh(geometry,material);mesh.name='BL_R01_'+kind;mesh.frustumCulled=false;mesh.castShadow=true;mesh.receiveShadow=true;mesh.customDepthMaterial=material.userData.depth;mesh.renderOrder=4;this.group.add(mesh);this.batches.push({kind,strands,mesh});}
 this.sampler=new BoundSurfaceSampler(g,this.batches.flatMap(b=>b.strands.flatMap(s=>s.anchors||[s.anchor])));this.revision++;this.lastSignature='';this.ready=true;this.lastGeometryMS=performance.now()-start;this.sync();this.onchange?.();}
 signature(){const g=this.viewer.mesh.geometry;return g.uuid+':'+g.attributes.position.version+':'+(this.viewer.mesh.morphTargetInfluences||[]).map(x=>(x||0).toFixed(6)).join(',');}
 sync(){if(this.disposed||!this.ready)return;const v=this.viewer,g=v.mesh.geometry,weights=v.mesh.morphTargetInfluences||[],s=this.settings;this.group.visible=s.enabled;const size=new THREE.Vector2();v.renderer.getDrawingBufferSize(size);for(const b of this.batches){b.mesh.visible=b.kind==='brow'?s.brows:s.lashes;b.mesh.material.userData.blViewport.copy(size);}
 const signature=this.signature();if(signature===this.lastSignature)return;this.lastSignature=signature;const start=performance.now();this.sampler.update(g,weights);let maxError=0;for(const b of this.batches){for(const strand of b.strands){const points=strand.anchors?strand.anchors.map(a=>a.sampled[0]):followLash(strand,this.frame);writeStrand(b.mesh.geometry,strand,points);strand.currentPoints=points;const root=strand.root.sampleCenter;maxError=Math.max(maxError,Math.hypot(...sub(points[0],root))*1000);}commitGeometry(b.mesh.geometry);}this.maxRootOffsetMM=maxError;this.lastUpdateMS=performance.now()-start;this.updateCount++;}
 set(patch){const next=validate({...this.settings,...patch}),geometryKeys=['seed','browDensity','browWidth','browLength','browArch','lashDensity','lashLength','lashCurl','fiberWidth','color'],rebuild=geometryKeys.some(k=>next[k]!==this.settings[k]);this.settings=next;if(rebuild)this.build();else{for(const b of this.batches)b.mesh.material.roughness=next.roughness;this.sync();}this.viewer.render();this.onchange?.();return this.report();}
 preset(id){if(!PRESETS[id])throw Error('未知眉睫预设');this.selected=id;this.settings=validate(PRESETS[id].values);this.build();this.viewer.render();return this.report();}
 archive(){return{schema:SCHEMA,settings:{...this.settings},selected:this.selected};}
 restore(a){if(a?.schema!==SCHEMA)throw Error('不支持的眉睫配方');const next=validate(a.settings);this.selected=a.selected||'custom';this.settings=next;this.build();this.viewer.render();}
 compare(on){this.settings.enabled=!on;this.sync();this.viewer.render();this.onchange?.();}
 view(name='eyes'){const f=this.frame,v=this.viewer;document.body.classList.remove('ear-focus','lips-focus');if(name==='body'){v.view('front');return;}let target=f.origin,d=.24,dx=0,dz=1;if(name==='face'){target=add(f.origin,mul(f.up,-.023));d=.46;}else if(name==='eye'){target=mix(this.landmark(39),this.landmark(36),.5);target=add(target,mul(f.up,.005));d=.115;}else if(name==='brow'){target=this.landmark(19);d=.105;}else if(name==='oblique'){dx=-.58;dz=.82;d=.255;}else if(name==='side'){dx=-.91;dz=.42;d=.235;}if(['eyes','oblique','side'].includes(name))target=add(target,mul(f.up,.006));v.camera.fov=31;v.camera.zoom=1;v.camera.up.set(...f.up);v.orbit.target.set(...target);v.camera.position.set(...add(target,add(mul(f.front,d*dz),mul(f.x,d*dx))));v.camera.updateProjectionMatrix();v.orbit.update();v.render();}
 report(){return{version:'BROW-LASH-R02',ready:this.ready,base:'HUMAN-INTEGRATED-R06 C3',coordinateContract:'native XYZ -> display X,Z,-Y, both charts and landmarks',selected:this.selected,settings:{...this.settings},revision:this.revision,drawBatches:this.batches.length,strands:Object.fromEntries(this.batches.map(b=>[b.kind,b.strands.length])),triangles:this.batches.reduce((n,b)=>n+b.mesh.geometry.index.count/3,0),geometryBytes:this.batches.reduce((n,b)=>n+Object.values(b.mesh.geometry.attributes).reduce((a,x)=>a+x.array.byteLength,0)+b.mesh.geometry.index.array.byteLength,0),maxRootOffsetMM:this.maxRootOffsetMM,buildMS:this.lastGeometryMS,updateMS:this.lastUpdateMS,updateCount:this.updateCount,msaa:this.msaa,coverage:'continuous blended per-fiber coverage; no sample dithering',bindingSampler:this.sampler?.report(),omitted:this.omitted,chartedTriangles:{display:this.display?.triangleCount,skin:this.skinChart?.triangleCount,globes:this.globeChart?.triangleCount},topology:this.model.canonical.topologySha256,focusedBlink:this.focusBlink?.report(),fiberScattering:'four optical paths / direct lights; approximate environment',nativeSkinUnmodified:true,externalCharacterImported:false,independentFilmBSDF:false,physicsSolver:false,errors:[...this.errors],userVisualAccepted:false};}
 dispose(){if(this.disposed)return;this.disposed=true;clearTimeout(this.editTimer);this.controller.removeEventListener('change',this.listener);this.viewer.render=this.originalRender;this.model.archive=this.originalArchive;this.model.restore=this.originalRestore;for(const b of this.batches){b.mesh.geometry.dispose();b.mesh.material.userData.depth.dispose();b.mesh.material.dispose();}this.group.removeFromParent();this.ready=false;}
}
