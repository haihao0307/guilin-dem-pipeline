function uniqueEdges(idx){const s=new Set(),out=[];for(let i=0;i<idx.length;i+=3)for(const [a,b]of[[idx[i],idx[i+1]],[idx[i+1],idx[i+2]],[idx[i+2],idx[i]]]){const x=Math.min(a,b),y=Math.max(a,b),k=x+','+y;if(!s.has(k)){s.add(k);out.push(x,y)}}return new Uint32Array(out)}

function partModel(mesh,time){
  let m=identity();
  const name=mesh.name;
  const omega=2*Math.PI/Math.max(.25,period);
  if(name.startsWith('dorsalFinlet')||name.startsWith('ventralFinlet')){
    const index=Number(name.match(/(\d+)$/)?.[1]??0);
    const side=name.startsWith('dorsal')?1:-1;
    const fm=finletMotion(index,time,{period,side});
    m=aroundPivot(mesh.centroid,rotateZ(fm.pitch));
    m=multiply(translate(0,fm.heave*side,0),m);
  }else if(name.startsWith('pectoral')){
    const sign=name.endsWith('Left')?1:-1;
    m=aroundPivot(mesh.centroid,rotateX(sign*(0.10+0.045*Math.sin(omega*time+sign*.4))));
  }else if(name.startsWith('pelvic')){
    const sign=name.endsWith('Left')?1:-1;
    m=aroundPivot(mesh.centroid,rotateX(sign*.045*Math.sin(omega*time-.5)));
  }else if(name.startsWith('caudal')){
    m=aroundPivot([-.47,-.05,0],rotateY(swimOn*.045*Math.sin(omega*time+.32)));
  }else if(name==='lowerLip'){
    m=aroundPivot([.486,-.027,0],rotateZ(-mouthOpen*.15));
  }else if(name==='mouthCavity'){
    m=multiply(translate(0,-mouthOpen*.0025,0),scale(1,1+mouthOpen*.22,1));
  }else if(name.startsWith('operculum')){
    const sign=name.endsWith('Left')?-1:1;
    const breathe=gillOpen*(.52+.48*Math.sin(time*1.7));
    m=translate(0,0,sign*.007*breathe);
  }
  return m;
}
