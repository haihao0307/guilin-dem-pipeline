const vs=`#version 300 es
precision highp float;
in vec3 position;
in vec3 normal;
in vec4 param;
uniform mat4 uViewProj;
uniform mat4 uModel;
uniform float uTime;
uniform float uAmplitude;
uniform float uPeriod;
uniform float uSwim;
out vec3 vNormal;
out vec3 vWorld;
out vec3 vLocal;
out vec4 vParam;
void main(){
  vec3 p=position;
  vec3 n=normal;
  float u=clamp(position.x+0.5,0.0,1.0);
  float tailness=clamp(1.0-u,0.0,1.0);
  float omega=6.28318530718/max(0.25,uPeriod);
  float phase=omega*uTime-1.18*tailness;
  float env=0.015+0.985*pow(tailness,2.65);
  float lateral=uSwim*uAmplitude*env*sin(phase);
  float dEnv=-2.65*pow(max(tailness,0.00001),1.65);
  float slope=uSwim*uAmplitude*(-dEnv*sin(phase)+1.18*env*cos(phase));
  float a=atan(slope),ca=cos(a),sa=sin(a);
  p.z+=lateral;
  n=normalize(vec3(ca*n.x+sa*n.z,n.y,-sa*n.x+ca*n.z));
  vec4 world=uModel*vec4(p,1.0);
  vWorld=world.xyz;
  vNormal=normalize(mat3(uModel)*n);
  vLocal=p;
  vParam=param;
  gl_Position=uViewProj*world;
}`;
